## What reserves mean

Reserves set money aside for future major repairs and replacements, separate from regular operating expenses. Both budget sheets reference the December 2022 reserve study. The newer study was completed in June 2026, after the 2026 budget had been prepared and the January budget meeting held. The expected approach is to use the June 2026 study to inform the 2027 budget, and future studies to inform the following year’s budget rather than routinely revising the current year’s budget. This is a planning approach, not an adopted policy or a decision on 2027 amounts.

(() => {
  const notes = [...document.querySelectorAll('.budget-note')];
  let active = null;
  let pinned = false;
  let timer;
  function close() {
    clearTimeout(timer);
    if (active) {
      active.querySelector('button').setAttribute('aria-expanded', 'false');
      active.querySelector('.budget-explanation').hidden = true;
    }
    active = null;
    pinned = false;
  }
  function show(note) {
    if (active !== note) close();
    active = note;
    const button = note.querySelector('button');
    const popup = note.querySelector('.budget-explanation');
    popup.hidden = false;
    button.setAttribute('aria-expanded', 'true');
    const rect = button.getBoundingClientRect();
    const box = popup.getBoundingClientRect();
    popup.style.left = Math.max(12, Math.min(rect.left, innerWidth - box.width - 12)) + 'px';
    popup.style.top = Math.max(12, rect.bottom + box.height + 10 > innerHeight ? rect.top - box.height - 8 : rect.bottom + 8) + 'px';
  }
  notes.forEach(note => {
    const button = note.querySelector('button');
    note.addEventListener('pointerenter', event => {
      clearTimeout(timer);
      if (event.pointerType === 'mouse' && !pinned) show(note);
    });
    note.addEventListener('pointerleave', event => {
      if (event.pointerType === 'mouse' && !pinned) timer = setTimeout(close, 180);
    });
    button.addEventListener('click', () => {
      if (active === note && pinned) close();
      else { show(note); pinned = true; }
    });
    button.addEventListener('keydown', event => {
      if (event.key === 'Escape') { close(); event.preventDefault(); }
    });
    note.addEventListener('focusout', event => {
      if (!note.contains(event.relatedTarget)) close();
    });
  });
  document.addEventListener('pointerdown', event => {
    if (active && !active.contains(event.target)) close();
  });
  document.addEventListener('keydown', event => { if (event.key === 'Escape') close(); });
  window.addEventListener('resize', close);
  window.addEventListener('scroll', close, true);
})();
</script>
